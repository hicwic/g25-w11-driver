cmake_minimum_required(VERSION 3.24)
project(g25_userspace VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(G25_BUILD_TOOL "Build the Windows HID CLI" ${WIN32})
option(G25_BUILD_DIRECTINPUT "Build the DirectInput force-feedback COM driver" ${WIN32})
option(G25_BUILD_TRAY "Build the Windows notification-area controller" ${WIN32})

add_library(g25_warnings INTERFACE)
if(MSVC)
  target_compile_options(g25_warnings INTERFACE /W4 /permissive- /utf-8 /EHsc)
else()
  target_compile_options(g25_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()

add_library(g25_effect_math src/directinput/effect_math.cpp)
target_include_directories(g25_effect_math PUBLIC src)
target_link_libraries(g25_effect_math PRIVATE g25_warnings)

if(WIN32)
  add_library(g25_settings src/settings/user_settings.cpp)
  target_include_directories(g25_settings PUBLIC src)
  target_compile_definitions(g25_settings PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(g25_settings PUBLIC advapi32 PRIVATE g25_warnings)
endif()

if(G25_BUILD_DIRECTINPUT)
  if(NOT WIN32)
    message(FATAL_ERROR "g25ff requires Windows")
  endif()
  add_library(g25ff SHARED
    src/directinput/g25_ff_driver.cpp
    src/directinput/g25_ff_driver.def
    src/device/hid_transport.cpp
    src/device/g25_device.cpp)
  target_compile_definitions(g25ff PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(g25ff PRIVATE g25_protocol g25_effect_math hid setupapi ole32 dinput8 dxguid g25_warnings)
  set_target_properties(g25ff PROPERTIES PREFIX "" OUTPUT_NAME "g25ff")
  if(MINGW)
    target_link_options(g25ff PRIVATE -static)
  endif()
  install(TARGETS g25ff RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
endif()

if(G25_BUILD_TRAY)
  if(NOT WIN32)
    message(FATAL_ERROR "g25tray requires Windows")
  endif()
  if(NOT G25_BUILD_TOOL)
    message(FATAL_ERROR "g25tray currently requires G25_BUILD_TOOL")
  endif()
  add_executable(g25tray WIN32 src/tray/main.cpp src/tray/g25tray.rc)
  target_compile_definitions(g25tray PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(g25tray PRIVATE g25_windows g25_settings shell32 g25_warnings)
  if(MINGW)
    target_link_options(g25tray PRIVATE -static)
  endif()
  install(TARGETS g25tray RUNTIME DESTINATION bin)
endif()

add_library(g25_protocol
  src/protocol/logitech_protocol.cpp
  src/protocol/force_feedback.cpp
  src/device/output_session.cpp)
target_include_directories(g25_protocol PUBLIC src)
target_link_libraries(g25_protocol PRIVATE g25_warnings)

if(G25_BUILD_TOOL)
  if(NOT WIN32)
    message(FATAL_ERROR "g25tool requires Windows; use -DG25_BUILD_TOOL=OFF for protocol tests")
  endif()
  add_library(g25_windows
    src/app/console_stop.cpp
    src/device/hid_transport.cpp
    src/device/g25_device.cpp
    src/device/directinput_probe.cpp)
  target_compile_definitions(g25_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(g25_windows PUBLIC g25_protocol hid setupapi dinput8 dxguid ole32 PRIVATE g25_warnings)
  add_executable(g25tool src/app/main.cpp)
  target_link_libraries(g25tool PRIVATE g25_windows g25_warnings)
  if(MINGW)
    target_link_options(g25tool PRIVATE -static)
  endif()
  install(TARGETS g25tool RUNTIME DESTINATION bin)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION share/g25tool)
  install(DIRECTORY docs/ DESTINATION share/g25tool/docs)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(effect_math_tests tests/effect_math_tests.cpp)
  target_link_libraries(effect_math_tests PRIVATE g25_effect_math g25_warnings)
  if(MINGW)
    target_link_options(effect_math_tests PRIVATE -static)
  endif()
  add_test(NAME effect_synthesis COMMAND effect_math_tests)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE g25_protocol g25_warnings)
  if(MINGW)
    target_link_options(protocol_tests PRIVATE -static)
  endif()
  add_test(NAME protocol_and_safety COMMAND protocol_tests)
  if(G25_BUILD_TOOL)
    add_executable(windows_tests tests/windows_tests.cpp)
    target_link_libraries(windows_tests PRIVATE g25_windows g25_settings g25_warnings)
    if(MINGW)
      target_link_options(windows_tests PRIVATE -static)
    endif()
    add_test(NAME windows_descriptor_and_cancel COMMAND windows_tests)
    add_test(NAME cli_dry_run_and_arguments
      COMMAND ${CMAKE_COMMAND} -DTOOL=$<TARGET_FILE:g25tool> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.cmake)
  endif()
  if(G25_BUILD_DIRECTINPUT)
    add_executable(com_driver_tests tests/com_driver_tests.cpp)
    target_compile_definitions(com_driver_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(com_driver_tests PRIVATE src)
    target_link_libraries(com_driver_tests PRIVATE ole32 dinput8 dxguid g25_warnings)
    if(MINGW)
      target_link_options(com_driver_tests PRIVATE -static)
    endif()
    add_test(NAME directinput_com_loading COMMAND com_driver_tests $<TARGET_FILE:g25ff>)
  endif()
endif()
